cmake_minimum_required(VERSION 3.16)
project(VocalHiFi VERSION 0.1.0 LANGUAGES CXX)
option(VP_BUILD_TESTS "Build Vocal Hi-Fi tests" ON)
add_library(vocal_hifi STATIC src/VocalProcessor.cpp src/vocal_hifi.cpp)
target_include_directories(vocal_hifi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(vocal_hifi PUBLIC cxx_std_17)
set_target_properties(vocal_hifi PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(vocal_hifi PRIVATE /W4 /permissive-)
else()
 target_compile_options(vocal_hifi PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(VP_BUILD_TESTS)
 enable_testing()
 add_executable(test_vocal_hifi tests/test_vocal_hifi.cpp)
 target_link_libraries(test_vocal_hifi PRIVATE vocal_hifi)
 add_test(NAME vocal_hifi_regression COMMAND test_vocal_hifi)
endif()
